cmake_minimum_required(VERSION 3.28)
project(Nu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
find_package(Threads REQUIRED)
add_library(nu_rules STATIC ../src/core/board.cpp)
target_include_directories(nu_rules PUBLIC ../include src)
target_compile_definitions(nu_rules PUBLIC GENSEKI_NU_CANCELLATION=1)
if(NOT MSVC)
  target_compile_options(nu_rules PUBLIC -std=c++2c -Wall -Wextra -Wpedantic)
endif()
add_executable(nu src/main.cpp)
target_link_libraries(nu PRIVATE nu_rules Threads::Threads)
add_executable(nu_tests tests/core.cpp)
target_link_libraries(nu_tests PRIVATE nu_rules Threads::Threads)
add_executable(nu_feature_bench tests/performance.cpp)
target_link_libraries(nu_feature_bench PRIVATE nu_rules Threads::Threads)
include(CTest)
add_test(NAME nu_core COMMAND nu_tests)
add_test(NAME nu_stress COMMAND nu stress 20)
add_test(NAME nu_perft COMMAND nu perft 3)
set_tests_properties(nu_perft PROPERTIES PASS_REGULAR_EXPRESSION "1440")
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME nu_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol.py $<TARGET_FILE:nu>)
endif()
