cmake_minimum_required(VERSION 3.28)
project(Nu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
find_package(Threads REQUIRED)
add_library(nu_rules STATIC ../src/core/board.cpp)
target_include_directories(nu_rules PUBLIC ../include src)
target_compile_definitions(nu_rules PUBLIC GENSEKI_NU_CANCELLATION=1)
if(NOT MSVC)
  target_compile_options(nu_rules PUBLIC -std=c++2c -Wall -Wextra -Wpedantic)
endif()
add_executable(nu src/main.cpp)
target_link_libraries(nu PRIVATE nu_rules Threads::Threads)
add_executable(nu_tests tests/core.cpp)
target_link_libraries(nu_tests PRIVATE nu_rules Threads::Threads)
add_executable(nu_symmetry_tests tests/symmetry.cpp)
target_link_libraries(nu_symmetry_tests PRIVATE nu_rules Threads::Threads)
add_executable(nu_fast_tests tests/fast.cpp)
target_link_libraries(nu_fast_tests PRIVATE nu_rules Threads::Threads)
add_executable(nu_feature_cost tests/feature_cost.cpp)
target_link_libraries(nu_feature_cost PRIVATE nu_rules Threads::Threads)
add_executable(nu_feature_bench tests/performance.cpp)
target_link_libraries(nu_feature_bench PRIVATE nu_rules Threads::Threads)
add_executable(nu_schema7_baseline tests/schema7_baseline.cpp)
target_link_libraries(nu_schema7_baseline PRIVATE nu_rules Threads::Threads)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  # This benchmark intentionally replaces both new and delete with counted malloc/free.
  target_compile_options(nu_schema7_baseline PRIVATE -Wno-mismatched-new-delete)
endif()
add_executable(nu_schema8_tests tests/schema8.cpp)
target_link_libraries(nu_schema8_tests PRIVATE nu_rules Threads::Threads)
add_executable(nu_campaign_cost tests/campaign_cost.cpp)
target_link_libraries(nu_campaign_cost PRIVATE nu_rules Threads::Threads)
add_executable(nu_hybrid_tests tests/hybrid.cpp)
target_link_libraries(nu_hybrid_tests PRIVATE nu_rules Threads::Threads)
add_executable(nu_hybrid_scores tests/hybrid_scores.cpp)
target_link_libraries(nu_hybrid_scores PRIVATE nu_rules Threads::Threads)
include(CTest)
add_test(NAME nu_core COMMAND nu_tests)
add_test(NAME nu_schema8 COMMAND nu_schema8_tests)
set_tests_properties(nu_schema8 PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
add_test(NAME nu_hybrid COMMAND nu_hybrid_tests)
set_tests_properties(nu_hybrid PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
add_test(NAME nu_fast COMMAND nu_fast_tests)
set_tests_properties(nu_fast PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
add_test(NAME nu_symmetry COMMAND nu_symmetry_tests)
set_tests_properties(nu_core nu_symmetry PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
add_test(NAME nu_stress COMMAND nu stress 20)
add_test(NAME nu_perft COMMAND nu perft 3)
set_tests_properties(nu_perft PROPERTIES PASS_REGULAR_EXPRESSION "1440")
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME nu_schema8_model COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/schema8_model.py)
  add_test(NAME nu_learning COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/learning.py)
  add_test(NAME nu_residual COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/residual.py)
  set_tests_properties(nu_learning nu_residual PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
  add_test(NAME nu_schema COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/schema.py)
  add_test(NAME nu_leakage COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/leakage.py)
  add_test(NAME nu_teacher COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/teacher.py)
  add_test(NAME nu_collection COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/collection.py)
  add_test(NAME nu_benchmark COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/benchmark.py)
  add_test(NAME nu_reencode COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reencode.py)
  add_test(NAME nu_arena_resume COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/symmetry_arena_resume.py)
  set_tests_properties(nu_schema PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
  set_tests_properties(nu_reencode PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
  set_tests_properties(nu_arena_resume PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
  add_test(NAME nu_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol.py $<TARGET_FILE:nu>)
endif()
