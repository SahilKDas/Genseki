cmake_minimum_required(VERSION 3.28)
project(Iota LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
add_library(iota_rules ../src/core/board.cpp)
target_include_directories(iota_rules PUBLIC ../include)
target_compile_definitions(iota_rules PUBLIC GENSEKI_NU_CANCELLATION)
add_executable(iota src/main.cpp)
target_link_libraries(iota PRIVATE iota_rules)
if(MINGW)
  target_link_options(iota PRIVATE -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_test(NAME iota_native COMMAND iota --selftest)
